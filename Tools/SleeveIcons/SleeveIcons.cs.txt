using System.Drawing;
using System.Drawing.Drawing2D;
using System.Drawing.Imaging;
using System.Drawing.Text;

var names=new[]{"sleeve_create","sleeve_manage","sleeve_cad_create","sleeve_opening","parameter_copy"};
var labels=new[]{"自動套管","套管管理","CAD 建立套管","建築開孔","參數複製"};
Directory.CreateDirectory(Path.Combine(AppContext.BaseDirectory,"icons"));
foreach(var name in names)foreach(int size in new[]{16,32}){
    using var bitmap=new Bitmap(size,size,PixelFormat.Format32bppArgb);
    using(var g=Graphics.FromImage(bitmap)){g.Clear(Color.Transparent);g.SmoothingMode=SmoothingMode.AntiAlias;g.ScaleTransform(size/32f,size/32f);Draw(g,name);}
    int opaque=0,clear=0;for(int x=0;x<size;x++)for(int y=0;y<size;y++){if(bitmap.GetPixel(x,y).A>0)opaque++;else clear++;}
    if(opaque<20||clear<20)throw new Exception("Invalid icon "+name);
    bitmap.Save(Path.Combine(AppContext.BaseDirectory,"icons",$"{name}_{size}.png"),ImageFormat.Png);
}
using(var sheet=new Bitmap(1060,350)){
using(var g=Graphics.FromImage(sheet)){
g.Clear(Color.White);g.TextRenderingHint=TextRenderingHint.ClearTypeGridFit;
using var dark=new SolidBrush(Color.FromArgb(49,56,64));g.FillRectangle(dark,0,175,1060,175);
using var font=new Font("Microsoft JhengHei UI",12);using var small=new Font("Microsoft JhengHei UI",9);
for(int row=0;row<2;row++)for(int i=0;i<names.Length;i++){
int x=20+i*208,y=row*175;using var text=new SolidBrush(row==0?Color.FromArgb(40,48,54):Color.White);
g.DrawString(labels[i],font,text,x,y+16);
using var icon32=new Bitmap(Path.Combine(AppContext.BaseDirectory,"icons",names[i]+"_32.png"));
using var icon16=new Bitmap(Path.Combine(AppContext.BaseDirectory,"icons",names[i]+"_16.png"));
g.DrawImageUnscaled(icon16,x+8,y+89);g.DrawImageUnscaled(icon32,x+48,y+81);
g.DrawString("16",small,text,x+6,y+122);g.DrawString("32",small,text,x+53,y+122);
var state=g.Save();g.TranslateTransform(x+103,y+60);g.ScaleTransform(2,2);g.SmoothingMode=SmoothingMode.AntiAlias;Draw(g,names[i]);g.Restore(state);
g.DrawString("放大檢視",small,text,x+107,y+136);
}}
sheet.Save(Path.Combine(AppContext.BaseDirectory,"icon-preview.png"),ImageFormat.Png);
}
Console.WriteLine("PASS 10 icons: correct dimensions, nonblank alpha content, transparent backgrounds. Light/dark preview generated.");

}}
